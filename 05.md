# ============================================================================
# MINI-RETO: ENTRENAMIENTO Q-LEARNING PARAMÉTRICO Y TELEMETRÍA COMPLETA
# Entorno: CliffWalking-v1 | Plataforma: Google Colab

import gymnasium as gym
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.animation as animation
from IPython.display import HTML
import random

# Fijar semillas para reproducibilidad
random.seed(42)
np.random.seed(42)

# ----------------------------------------------------------------------------
# 1. PARÁMETROS EXPERIMENTALES DEL ESTUDIANTE (MODIFICAR AQUÍ EN CADA CORRIDA)
# ----------------------------------------------------------------------------
# Modifica estas 4 variables para registrar cada fila de la TABLA DEL RETO:
#   Corrida 1 (Base):           alpha = 0.10, gamma = 0.95, epsilon = 0.20, episodios = 400
#   Corrida 2 (Miope):          alpha = 0.10, gamma = 0.15, epsilon = 0.20, episodios = 400
#   Corrida 3 (Sin Explorar):   alpha = 0.10, gamma = 0.95, epsilon = 0.00, episodios = 400
#   Corrida 4 (Infraentrenado): alpha = 0.10, gamma = 0.95, epsilon = 0.20, episodios = 30

alpha = 0.10          # Tasa de aprendizaje
gamma = 0.95          # Factor de descuento temporal
epsilon = 0.20        # Probabilidad de exploración Epsilon-Greedy
episodios = 400       # Ciclos totales de entrenamiento

# ----------------------------------------------------------------------------
# 2. BUCLE DE ENTRENAMIENTO Q-LEARNING CON MONITOREO DE COSTO
# ----------------------------------------------------------------------------
env = gym.make('CliffWalking-v1')
n_estados = env.observation_space.n
n_acciones = env.action_space.n

Q = np.zeros((n_estados, n_acciones))

historial_recompensas = []
historial_pasos = []
historial_error_td = []

for ep in range(episodios):
    s, _ = env.reset()
    recompensa_ep = 0.0
    pasos_ep = 0
    errores_ep = []
    
    while True:
        pasos_ep += 1
        
        # Selección de acción: Epsilon-Greedy
        if random.random() < epsilon:
            a = env.action_space.sample()
        else:
            a = np.argmax(Q[s, :])
            
        s_sig, r, term, trunc, _ = env.step(a)
        
        # Actualización de Bellman TD
        td_target = r + gamma * np.max(Q[s_sig, :])
        td_error = td_target - Q[s, a]
        Q[s, a] += alpha * td_error
        
        errores_ep.append(abs(td_error))
        recompensa_ep += r
        s = s_sig
        
        if term or trunc:
            break
            
    historial_recompensas.append(recompensa_ep)
    historial_pasos.append(pasos_ep)
    historial_error_td.append(np.mean(errores_ep))

# ----------------------------------------------------------------------------
# 3. EVALUACIÓN DETERMINISTA DE LA POLÍTICA APRENDIDA (EXPLOTACIÓN PURA)
# ----------------------------------------------------------------------------
s_eval, _ = env.reset(seed=42)
trayectoria_final = [s_eval]
retorno_eval = 0.0
max_pasos_eval = 35

for _ in range(max_pasos_eval):
    a_opt = np.argmax(Q[s_eval, :])
    s_eval, r_eval, term, trunc, _ = env.step(a_opt)
    trayectoria_final.append(s_eval)
    retorno_eval += r_eval
    if term or trunc:
        break

total_pasos = len(trayectoria_final) - 1
llego_meta = (trayectoria_final[-1] == 47)
error_td_final = np.mean(historial_error_td[-10:]) if len(historial_error_td) >= 10 else np.mean(historial_error_td)

# ----------------------------------------------------------------------------
# 4. REPORTE EN CONSOLA (TRANSCRIBIR DIRECTAMENTE A LA TABLA)
# ----------------------------------------------------------------------------
print("=" * 80)
print(f"{'TELEMETRÍA PARA COMPLETAR LA TABLA DEL RETO':^80}")
print("=" * 80)
print(f" -> Tasa de Aprendizaje (alpha):       {alpha:.2f}")
print(f" -> Descuento Temporal (gamma):         {gamma:.2f}")
print(f" -> Tasa de Exploración (epsilon):     {epsilon:.2f}")
print(f" -> Volumen de Episodios (N):          {episodios}")
print(f" -> Retorno Final Obtenido (G0):       {retorno_eval:+06.1f} pts")
print(f" -> ¿Alcanzó la Meta Física?:          {'SÍ' if llego_meta else 'NO'}")
print(f" -> Pasos Empleados por el Robot:      {total_pasos} pasos")
print(f" -> Error TD Final Promedio (|delta|): {error_td_final:.4f}")
print("=" * 80)

# ----------------------------------------------------------------------------
# 5. GRÁFICAS DE APRENDIZAJE: RECOMPENSA ACUMULADA Y ERROR TD
# ----------------------------------------------------------------------------
ventana = min(15, max(1, episodios // 5))
media_movil_rew = np.convolve(historial_recompensas, np.ones(ventana)/ventana, mode='valid')
media_movil_error = np.convolve(historial_error_td, np.ones(ventana)/ventana, mode='valid')

fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(13, 4.2), dpi=110)

# Subplot 1: Curva de Recompensa
ax1.plot(historial_recompensas, color='#94a3b8', alpha=0.35, label='Recompensa bruta')
ax1.plot(range(ventana - 1, episodios), media_movil_rew, color='#16a34a', lw=2, label=f'Media móvil ({ventana} ep)')
ax1.axhline(-13.0, color='#dc2626', linestyle='--', lw=1.5, label='Óptimo (-13.0 pts)')
ax1.set_title(f'Recompensa Acumulada (alpha={alpha}, gamma={gamma})', fontweight='bold')
ax1.set_xlabel('Episodio')
ax1.set_ylabel('Retorno G0')
ax1.legend(loc='lower right', frameon=True)
ax1.grid(True, linestyle='--', alpha=0.3)

# Subplot 2: Curva de Costo (Error TD)
ax2.plot(historial_error_td, color='#f59e0b', alpha=0.35, label='Error TD bruto')
ax2.plot(range(ventana - 1, episodios), media_movil_error, color='#b45309', lw=2, label='Media móvil error')
ax2.set_title(f'Costo: Magnitud del Error TD (|delta| = {error_td_final:.4f})', fontweight='bold')
ax2.set_xlabel('Episodio')
ax2.set_ylabel('Error TD Promedio')
ax2.legend(loc='upper right', frameon=True)
ax2.grid(True, linestyle='--', alpha=0.3)

plt.tight_layout()
plt.show()

# ----------------------------------------------------------------------------
# 6. ANIMACIÓN INTERACTIVA HTML5 DE LA POLÍTICA FINAL
# ----------------------------------------------------------------------------
def generar_animacion(ruta, llego):
    fig_anim, ax = plt.subplots(figsize=(10, 3.8), dpi=110)
    
    def dibujar_frame(i):
        ax.clear()
        ax.set_xlim(-0.5, 11.5)
        ax.set_ylim(3.5, -0.5)
        ax.set_xticks(range(12))
        ax.set_yticks(range(4))
        ax.grid(True, color='gray', linestyle='--', alpha=0.4)
        
        ax.add_patch(plt.Rectangle((0.5, 2.5), 10, 1, color='#ef4444', alpha=0.45))
        ax.text(5.5, 3, "ACANTILADO (-100 pts)", ha='center', va='center', fontweight='bold', color='#7f1d1d')
        
        ax.add_patch(plt.Rectangle((-0.5, 2.5), 1, 1, color='#94a3b8', alpha=0.35))
        ax.text(0, 3, "INICIO\n(s=36)", ha='center', va='center', fontweight='bold', color='#1e293b', fontsize=8)
        
        ax.add_patch(plt.Rectangle((10.5, 2.5), 1, 1, color='#22c55e', alpha=0.45))
        ax.text(11, 3, "META\n(s=47)", ha='center', va='center', fontweight='bold', color='#14532d', fontsize=8)
        
        for k in range(i):
            s1, s2 = ruta[k], ruta[k+1]
            ax.plot([s1 % 12, s2 % 12], [s1 // 12, s2 // 12], color='#0284c7', linestyle=':', lw=2, marker='.')

        s_act = ruta[i]
        r_act, c_act = s_act // 12, s_act % 12
        es_fin = (i == len(ruta) - 1)
        color_robot = ('#16a34a' if llego else '#dc2626') if es_fin else '#0284c7'
        
        ax.add_patch(plt.Circle((c_act, r_act), 0.35, color=color_robot, ec='black', lw=1.5))
        ax.text(c_act, r_act, "ROBOT", ha='center', va='center', color='white', fontweight='bold', fontsize=7)
        ax.set_title(f"Paso {i} de {len(ruta)-1} | Estado: s = {s_act:02d} | {'ÉXITO' if (es_fin and llego) else ('CAÍDA' if (es_fin and not llego) else 'EN TRÁNSITO')}", fontweight='bold')

    anim = animation.FuncAnimation(fig_anim, dibujar_frame, frames=len(ruta), interval=400, repeat=False)
    plt.close()
    return HTML(anim.to_jshtml())

generar_animacion(trayectoria_final, llego_meta)

# ============================================================================
# MINI-RETO: ENTRENAMIENTO Q-LEARNING CON INGENIERÍA DE RECOMPENSAS
# Entorno: CliffWalking-v1 | Plataforma: Google Colab
# ============================================================================

import gymnasium as gym
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.animation as animation
from IPython.display import HTML
import random

# Semillas para reproducibilidad experimental
random.seed(42)
np.random.seed(42)

# ----------------------------------------------------------------------------
# 1. PARÁMETROS EXPERIMENTALES DEL ESTUDIANTE (MODIFICAR AQUÍ)
# ----------------------------------------------------------------------------
# Hiperparámetros de aprendizaje
alpha = 0.10          # Tasa de aprendizaje (Learning Rate)
gamma = 0.95          # Factor de descuento temporal
epsilon = 0.20        # Probabilidad de exploración Epsilon-Greedy
episodios = 400       # Número total de episodios de entrenamiento

# --- INGENIERÍA DE RECOMPENSAS (MODIFICAR PARA EXPERIMENTAR) ---
r_paso = 0         # Costo energético de batería por paso (Nominal: -1.0)
r_caida = -100.0      # Penalización por precipitarse al vacío (Nominal: -100.0)

# ----------------------------------------------------------------------------
# 2. BUCLE DE ENTRENAMIENTO Q-LEARNING CON INTERCEPTACIÓN DE RECOMPENSAS
# ----------------------------------------------------------------------------
env = gym.make('CliffWalking-v1')
n_estados = env.observation_space.n
n_acciones = env.action_space.n

# Inicializar tabla de valor-acción en ceros
Q = np.zeros((n_estados, n_acciones))

historial_recompensas = []
historial_pasos = []
historial_error_td = []

for ep in range(episodios):
    s, _ = env.reset()
    recompensa_ep = 0.0
    pasos_ep = 0
    errores_ep = []

    while True:
        pasos_ep += 1

        # Selección de acción: Epsilon-Greedy
        if random.random() < epsilon:
            a = env.action_space.sample()        # Exploración aleatoria
        else:
            a = np.argmax(Q[s, :])               # Explotación voraz

        s_sig, r_env, term, trunc, _ = env.step(a)

        # Interceptación de la señal de refuerzo física
        if r_env == -1.0:
            r = r_paso
        elif r_env == -100.0:
            r = r_caida
        else:
            r = r_env

        # Ecuación Maestra de Bellman TD (Q-Learning)
        td_target = r + gamma * np.max(Q[s_sig, :])
        td_error = td_target - Q[s, a]
        Q[s, a] += alpha * td_error

        # Registro de métricas inmediatas
        errores_ep.append(abs(td_error))
        recompensa_ep += r
        s = s_sig

        if term or trunc:
            break

    historial_recompensas.append(recompensa_ep)
    historial_pasos.append(pasos_ep)
    historial_error_td.append(np.mean(errores_ep))

# ----------------------------------------------------------------------------
# 3. EVALUACIÓN DETERMINISTA DE LA POLÍTICA APRENDIDA (MODO EXPLOTACIÓN PURA)
# ----------------------------------------------------------------------------

trayectoria_final = [s_eval]
retorno_eval = 0.0
max_pasos_eval = 40

for _ in range(max_pasos_eval):
    a_opt = np.argmax(Q[s_eval, :])              # Explotación estricta (epsilon = 0)
    s_eval, r_env, term, trunc, _ = env.step(a_opt)

    # Aplicar la misma recompensa personalizada en evaluación
    r_eval = r_caida if r_env == -100.0 else (r_paso if r_env == -1.0 else r_env)

    trayectoria_final.append(s_eval)
    retorno_eval += r_eval

    if term or trunc:
        break

total_pasos = len(trayectoria_final) - 1
llego_meta = (trayectoria_final[-1] == 47)
error_td_final = np.mean(historial_error_td[-15:]) if len(historial_error_td) >= 15 else np.mean(historial_error_td)

print(f" -> Factor de Descuento (gamma):        {gamma:.2f}")
print(f" -> Tasa de Exploración (epsilon):     {epsilon:.2f}")
print(f" -> Volumen de Episodios (N):          {episodios}")
print(f" -> Costo por Paso (r_paso):           {r_paso:+.2f} pts")
print(f" -> Castigo por Caída (r_caida):       {r_caida:+.2f} pts")
print(f" -> Retorno Final Acumulado (G0):      {retorno_eval:+06.1f} pts")
print(f" -> ¿Alcanzó la Meta Física (s=47)?:   {'SÍ' if llego_meta else 'NO'}")
print(f" -> Pasos Empleados por el Robot:      {total_pasos} pasos")
print(f" -> Error TD Final Promedio (|delta|): {error_td_final:.4f}")
print("=" * 82)

# ----------------------------------------------------------------------------
# 5. GRÁFICAS DE EVALUACIÓN: RECOMPENSA ACUMULADA Y ERROR TD
# ----------------------------------------------------------------------------
ventana = min(15, max(1, episodios // 8))
media_movil_rew = np.convolve(historial_recompensas, np.ones(ventana)/ventana, mode='valid')
media_movil_error = np.convolve(historial_error_td, np.ones(ventana)/ventana, mode='valid')

ax1.plot(historial_recompensas, color='#94a3b8', alpha=0.35, label='Recompensa bruta')
ax1.plot(range(ventana - 1, episodios), media_movil_rew, color='#16a34a', lw=2, label=f'Media móvil ({ventana} ep)')
ax1.set_title('Evolución de la Recompensa Acumulada (Retorno G0)', fontweight='bold')
ax1.set_xlabel('Episodio de Entrenamiento')
ax1.set_ylabel('Retorno Acumulado')
ax1.legend(loc='lower right', frameon=True)
ax1.grid(True, linestyle='--', alpha=0.4)

# Gráfica 2: Curva de Costo / Error TD
ax2.plot(historial_error_td, color='#f59e0b', alpha=0.35, label='Error TD bruto')
ax2.plot(range(ventana - 1, episodios), media_movil_error, color='#b45309', lw=2, label=f'Media móvil error ({ventana} ep)')
ax2.set_title(f'Función de Costo: Error TD (|delta| = {error_td_final:.4f})', fontweight='bold')
ax2.set_xlabel('Episodio de Entrenamiento')
ax2.set_ylabel('Magnitud del Error TD')
ax2.legend(loc='upper right', frameon=True)
ax2.grid(True, linestyle='--', alpha=0.4)
