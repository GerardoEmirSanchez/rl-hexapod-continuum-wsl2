```Bash
#!/usr/bin/env python3
"""
==============================================================================
MR3005C: Sistemas Ciberfísicos — Módulo 9: Aprendizaje por Refuerzo
Sesión 5: Q-Learning Model-Free en Gymnasium (CliffWalking-v1)
==============================================================================
"""

import gymnasium as gym
import numpy as np
import random
import sys

# Semillas estocásticas para consistencia de cátedra
random.seed(42)
np.random.seed(42)

def main():
    print("\n" + "="*75)
    print(" MR3005C: APRENDIZAJE Q-LEARNING EN VIVO (SESIÓN 5)")
    print(" Entorno: CliffWalking-v1 | Plataforma Industrial Gymnasium")
    print("="*75 + "\n")

    # 1. Instanciación del entorno estándar
    env = gym.make('CliffWalking-v1')
    n_estados = env.observation_space.n
    n_acciones = env.action_space.n

    # 2. Memoria tabular inicializada en ceros (Optimismo inicial)
    Q = np.zeros((n_estados, n_acciones))

    # 3. Hiperparámetros base de aprendizaje
    alpha = 0.10          # Tasa de aprendizaje
    gamma = 0.95          # Factor de descuento temporal
    epsilon = 0.20        # Exploración Epsilon-Greedy
    episodios = 400       # Ciclos de entrenamiento

    print(f"[INFO] Espacio de Estados: {n_estados} | Espacio de Acciones: {n_acciones}")
    print(f"[INFO] Entrenando durante {episodios} episodios...")

    # Bucle de entrenamiento por Diferencia Temporal (Q-Learning)
    for ep in range(1, episodios + 1):
        s, _ = env.reset()
        terminado = False
        truncado = False

        while not (terminado or truncado):
            # Selección Epsilon-Greedy
            if random.random() < epsilon:
                a = env.action_space.sample()
            else:
                a = np.argmax(Q[s, :])

            s_sig, r, terminado, truncado, _ = env.step(a)

            # Ecuación Maestra de Bellman TD
            td_target = r + gamma * np.max(Q[s_sig, :])
            td_error = td_target - Q[s, a]
            Q[s, a] += alpha * td_error

            s = s_sig

        if ep % 100 == 0:
            print(f" -> Episodio {ep:03d}/{episodios} completado.")

    print("\n" + "="*75)
    print(" EVALUACIÓN DE LA POLÍTICA APRENDIDA (MODO EXPLOTACIÓN PURA)")
    print("="*75)

    # Evaluación determinista (epsilon = 0)
    s_eval, _ = env.reset(seed=42)
    ruta = [s_eval]
    retorno_total = 0.0
    pasos = 0

    while True:
        pasos += 1
        a_opt = np.argmax(Q[s_eval, :])
        s_eval, r_eval, term_eval, trunc_eval, _ = env.step(a_opt)
        ruta.append(s_eval)
        retorno_total += r_eval

        if term_eval or trunc_eval or pasos > 30:
            break

    print(f" -> Total de Pasos Empleados: {len(ruta) - 1} pasos")
    print(f" -> Retorno Total Acumulado:  {retorno_total:.1f} puntos")
    print(f" -> ¿Alcanzó la Meta (s=47)?: {'SÍ' if ruta[-1] == 47 else 'NO'}")
    print(f" -> Secuencia de Estados:    {ruta}")
    print("="*75 + "\n")

if __name__ == '__main__':
    main()
```

## 4. Solución Oficial del Mini-Reto para el Profesor (`M9_s5_mini_reto_resuelto.py`)

Guarda este archivo en tu estación de trabajo para proyectar o contrastar las variaciones de hiperparámetros con los estudiantes al concluir los 15 minutos[cite: 10]:

```Bash
cat << 'EOF' > ~/rl_ws/M9/M9_s5_mini_reto_resuelto.py
#!/usr/bin/env python3
"""
==============================================================================
MR3005C: Sistemas Ciberfísicos — Módulo 9: Aprendizaje por Refuerzo
Sesión 5: Solución Oficial del Mini-Reto de Estrés de Hiperparámetros
==============================================================================
"""

import gymnasium as gym
import numpy as np
import random

def entrenar_y_evaluar(alpha_val, gamma_val, nombre_prueba):
    random.seed(42)
    np.random.seed(42)

    env = gym.make('CliffWalking-v1')
    Q = np.zeros((env.observation_space.n, env.action_space.n))
    epsilon = 0.20
    episodios = 350

    for _ in range(episodios):
        s, _ = env.reset()
        while True:
            a = env.action_space.sample() if random.random() < epsilon else np.argmax(Q[s, :])
            s_sig, r, term, trunc, _ = env.step(a)
            Q[s, a] += alpha_val * (r + gamma_val * np.max(Q[s_sig, :]) - Q[s, a])
            s = s_sig
            if term or trunc:
                break

    # Explotación voraz
    s_eval, _ = env.reset(seed=42)
    pasos = 0
    retorno = 0.0
    while True:
        pasos += 1
        a_opt = np.argmax(Q[s_eval, :])
        s_eval, r_eval, term_eval, trunc_eval, _ = env.step(a_opt)
        retorno += r_eval
        if term_eval or trunc_eval or pasos >= 30:
            break

    exito = (s_eval == 47)
    print(f"| {nombre_prueba:<22} | alpha={alpha_val:<4} | gamma={gamma_val:<4} | Pasos: {pasos:02d} | Retorno: {retorno:+06.1f} | Meta: {'SÍ' if exito else 'NO'} |")

def main():
    print("\n" + "="*80)
    print(" RESULTADOS COMPARATIVOS DEL MINI-RETO DE HIPERPARÁMETROS")
    print("="*80)
    entrenar_y_evaluar(0.10, 0.95, "Config Base (Óptima)")
    entrenar_y_evaluar(0.10, 0.10, "Reto 1 (Agente Miope)")
    entrenar_y_evaluar(0.95, 0.95, "Reto 2 (Sobre-escritura)")
    print("="*80)
    print("\nConclusiones Técnicas:")
    print("1. Con gamma = 0.10, la señal decae como (0.1)^13 = 1e-13; el agente ignora la meta lejana.")
    print("2. Con alpha = 0.95, la memoria borra la experiencia previa ante cualquier paso ruidoso.")
    print("="*80 + "\n")

if __name__ == '__main__':
    main()
```
